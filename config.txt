# Default Nerves RPi 3 config.txt
#
# It's possible to override this file by using a custom fwup.conf
# configuration to pull in a replacement.
#
# Useful links:
# https://www.raspberrypi.org/documentation/configuration/config-txt/README.md
# https://www.raspberrypi.org/documentation/configuration/device-tree.md
# https://github.com/raspberrypi/documentation/blob/master/configuration/device-tree.md
# https://github.com/raspberrypi/firmware/blob/master/boot/overlays/README

kernel=zImage

# Disable the boot rainbow
disable_splash=1

# This, along with the Raspberry Pi "x" firmware is needed for the camera
# to work. The Raspberry Pi "x" firmware is selected via the Buildroot
# configuration. See Target packages->Hardware handling->Firmware.
gpu_mem=192

# Enable I2C, SPI, and audio
dtparam=i2c_arm=on
dtparam=spi=on
dtparam=audio=on

# Automatically load overlays for detected cameras
camera_auto_detect=1

# Raspberry Pi 7" DSI Touchscreen — legacy fbdev path.
# Under VC4 KMS the DSI subdevice doesn't bind on this firmware/kernel
# combo, so we let the firmware (start.elf) drive the DSI panel
# directly. The kernel sees /dev/fb0; Qt5 WebEngine's linuxfb QPA
# plugin renders to it.
dtoverlay=rpi-ft5406
dtoverlay=rpi-backlight
disable_overscan=1

# Mirror the DSI panel output to HDMI so a TV/monitor plugged into
# the Pi shows the same kiosk UI. `max_framebuffers=2` makes the
# firmware allocate /dev/fb1 for the HDMI display in addition to
# /dev/fb0 for the DSI panel. `hdmi_force_hotplug=1` keeps the HDMI
# output enabled even when no display is attached at boot (so a
# user can hot-plug a monitor later). The mirroring itself is done
# by `BabyFirmware.FbMirror`, which blits /dev/fb0 → /dev/fb1 at
# ~30 fps.
max_framebuffers=2
hdmi_force_hotplug=1
hdmi_group=1
hdmi_mode=4
hdmi_drive=2

# Comment this in or modify to enable OneWire
# NOTE: check that the overlay that you specify is in the boot partition or
#       this won't work.
#dtoverlay=w1-gpio-pullup,gpiopin=4

# The ramoops overlay works with the pstore driver to preserve crash
# information across reboots in DRAM
dtoverlay=ramoops

# Enable the UART (/dev/ttyAMA0) on the RPi3.
enable_uart=1
dtoverlay=miniuart-bt
# Set the GPU frequency so that the MiniUART (/dev/ttyS0) can be used for
# Bluetooth. This has a small performance impact. See
# https://www.raspberrypi.org/documentation/computers/config_txt.html#overclocking.
core_freq=250
